cmake_minimum_required(VERSION 3.20)
project(PrismCurrents VERSION 1.18.0 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJC OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
set(USE_AUDIO OFF CACHE BOOL "" FORCE)
set(SUPPORT_MODULE_RAUDIO OFF CACHE BOOL "" FORCE)
set(OPENGL_VERSION "3.3" CACHE STRING "" FORCE)
add_subdirectory(vendor/raylib EXCLUDE_FROM_ALL)
get_target_property(raylib_definitions raylib COMPILE_DEFINITIONS)
list(FILTER raylib_definitions EXCLUDE REGEX "^MAX_MESH_VERTEX_BUFFERS=")
set_property(TARGET raylib PROPERTY COMPILE_DEFINITIONS "${raylib_definitions}")
target_compile_definitions(raylib PRIVATE MAX_MESH_VERTEX_BUFFERS=9)

add_library(prism_audio src/audio.cpp)
target_include_directories(prism_audio PUBLIC src PRIVATE vendor/raylib/src/external)
if(APPLE)
  target_sources(prism_audio PRIVATE src/native.mm)
  target_link_libraries(prism_audio PUBLIC "-framework CoreAudio" "-framework AudioToolbox" "-framework AudioUnit" "-framework AVFoundation" "-framework AppKit" "-framework UniformTypeIdentifiers")
  add_executable(PrismCurrents MACOSX_BUNDLE src/main.cpp src/renderer.cpp src/patch.cpp)
  target_sources(PrismCurrents PRIVATE src/webcam.mm)
  set_source_files_properties(src/webcam.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  set_target_properties(PrismCurrents PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
    MACOSX_BUNDLE_BUNDLE_NAME "Prism Currents"
    MACOSX_BUNDLE_GUI_IDENTIFIER "studio.prismcurrents.visualiser"
    MACOSX_BUNDLE_BUNDLE_VERSION "1.18.0"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "1.18.0")
  target_link_libraries(PrismCurrents PRIVATE "-framework OpenGL" "-framework CoreMedia" "-framework CoreVideo")
else()
  target_sources(prism_audio PRIVATE src/native_stub.cpp)
  target_link_libraries(prism_audio PUBLIC ${CMAKE_DL_LIBS})
  if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(prism_audio PUBLIC Threads::Threads m)
  endif()
  add_executable(PrismCurrents src/main.cpp src/renderer.cpp src/patch.cpp)
endif()
target_link_libraries(PrismCurrents PRIVATE prism_audio raylib)
target_include_directories(PrismCurrents PRIVATE vendor/raylib/src/external)
target_include_directories(PrismCurrents SYSTEM PRIVATE vendor/raylib/src)
target_compile_options(PrismCurrents PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(PrismCurrents PRIVATE PRISM_SOURCE_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
if(APPLE)
  set(ASSET_DEST "$<TARGET_FILE_DIR:PrismCurrents>/../Resources")
else()
  set(ASSET_DEST "$<TARGET_FILE_DIR:PrismCurrents>")
endif()
add_custom_command(TARGET PrismCurrents POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${ASSET_DEST}/licenses"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${ASSET_DEST}/assets"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY.md" "${ASSET_DEST}/licenses/THIRD_PARTY.md"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib/LICENSE" "${ASSET_DEST}/licenses/raylib-LICENSE")
include(CTest)
if(BUILD_TESTING)
  add_executable(prism_tests tests/audio_tests.cpp)
  target_include_directories(prism_tests PRIVATE vendor/raylib/src/external)
  target_link_libraries(prism_tests PRIVATE prism_audio)
  add_test(NAME audio_analysis_and_decoder COMMAND prism_tests)
  add_executable(prism_response_tests tests/response_tests.cpp)
  target_link_libraries(prism_response_tests PRIVATE prism_audio)
  add_test(NAME musical_response COMMAND prism_response_tests)
  add_executable(prism_evolution_tests tests/evolution_tests.cpp)
  target_include_directories(prism_evolution_tests PRIVATE src)
  add_test(NAME evolving_form COMMAND prism_evolution_tests)
endif()
install(TARGETS PrismCurrents BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(NOT APPLE)
  install(DIRECTORY assets DESTINATION bin)
endif()
